% 3 ---------------------------------------------------------------
\begin{frame}{Three claims}
  \begin{enumerate}
    \setlength\itemsep{1em}
    \item \key{An unremunerated digital euro hardens the lower bound.}\\
      It is cash without storage costs: on capped balances the deposit-rate floor moves from $-c_{\text{cash}}$ back to $0$.
      \muted{(Known in policy work; I give a one-line formalisation.)}
    \item \key{If a holding cost is programmed, it must be demurrage, not expiry.}\\
      Expiry is a fiscal-programme tool; only continuous decay is an interest rate.
      \muted{(Model result.)}
    \item \key{Remuneration flexibility protects policy space; it does not create it.}\\
      While cash exists, a negative digital-euro rate cannot go below the cash floor.
      \muted{Policy: keep zero as the default, legislate a contingent rate clause.}
  \end{enumerate}
\end{frame}

\begin{frame}{Why programmable money tempts policymakers}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \textbf{The macro problem}
      \begin{itemize}
        \item Liquidity trap: $MV = PY$, but $V$ falls when $M$ rises
        \item Rates stuck at the effective lower bound (ECB DFR $-0.50\%$, 2019--22)
        \item Desire to bypass banks and push spending directly
      \end{itemize}
    \end{column}
    \begin{column}{0.52\textwidth}
      \textbf{Expiry works, empirically}
      \begin{itemize}
        \item China digital coupons: \textbf{\textyen 3} extra spending per \textyen 1 of subsidy\\
              \muted{\footnotesize Xing et al.\ (AEJ:Macro 2023); Ding et al.\ (2024): 3.1--3.3}
        \item France RCT: one-month MPC \textbf{61\%} with a 3-week expiry vs \textbf{23\%} without\\
              \muted{\footnotesize Boehm, Fize \& Jaravel (2024)}
        \item Japan 1999 coupons: weak effect\\
              \muted{\footnotesize Hsieh, Shimizutani \& Hori (JPubE 2010)}
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \begin{center}
    \key{Question: can the digital euro itself carry this ``use it or lose it'' force?}
  \end{center}
\end{frame}

% 6 ---------------------------------------------------------------
\begin{frame}{Central-bank money is the deposit's outside option}
  A depositor can always switch to a central-bank liability $j$ with rate $i_j$ and holding cost $c_j$.
  Banks cannot push the deposit rate below the best outside option:
  \[
    r_D \;\ge\; \max_j \,\bigl\{\, i_j - c_j \,\bigr\}
    \qquad \text{(on balances that option can absorb)}
  \]
  \begin{center}
  \small
  \begin{tabular}{@{}lccc@{}}
    \toprule
    Outside option & $i_j$ & $c_j$ & Floor on $r_D$ \\
    \midrule
    Cash & $0$ & $c_{\text{cash}}>0$ \muted{(storage, insurance)} & $-c_{\text{cash}}$ \\
    Digital euro, Art.\ 16(8) & $0$ & $\approx 0$ & \alert{$0$}, up to the limit $\bar M$ \\
    Digital euro at $-\lambda$ & $-\lambda$ & $\approx 0$ & $\max\{-\lambda,\,-c_{\text{cash}}\}$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.5em}
  \begin{itemize}
    \item The \textbf{ZLB} and \textbf{disintermediation} are the same mechanism seen from two sides
    \item The holding limit $\bar M$ decides \emph{how much} money enjoys the zero floor \muted{(companion talk)}
  \end{itemize}
\end{frame}

% 7 ---------------------------------------------------------------
\begin{frame}{Claim 1: a zero-rate digital euro is frictionless cash}
  \begin{itemize}
    \setlength\itemsep{0.7em}
    \item 2019--22: the ECB went to $-0.50\%$ without a run into cash, because hoarding notes is costly
    \item A digital euro at $0$ removes that cost on the first $\bar M$ per person
    \item Consequences when rates are negative:
      \begin{itemize}
        \item deposits up to $\bar M$ flow out exactly when banks are weakest
        \item pass-through of negative rates to retail balances is capped at $0$
        \item disintermediation pressure peaks in the \textbf{negative-rate} regime, not in normal times
      \end{itemize}
    \item So zero remuneration and the holding limit are a \key{package}: the limit is what makes the zero floor affordable
  \end{itemize}
  \vspace{0.5em}
  \muted{\small Consistent with ECB (2020) and Pirgmann \& Wawrosz (2024). New here: the floor formula and the link to $\bar M$.}
\end{frame}

% 8 ---------------------------------------------------------------
\begin{frame}{Claim 2: a model of programmable holding costs}
  \begin{itemize}
    \item Token value per unit: $V$. Useful purchases are worth $1$; holders search with intensity $\kappa$ at cost $\kappa^2/2a$
    \item A \textbf{junk outlet} (low-value goods, dumping on a secondary market) is always available at $\phi<1$
    \item Discount rate $r$; velocity is $\kappa$
  \end{itemize}
  \vspace{0.3em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Hard expiry at $T$ \quad ($\tau = T-t$)}
        \[
          \frac{dV}{d\tau} = -rV + \frac{a}{2}(1-V)^2,\quad V(0)=\phi
        \]
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Demurrage at rate $\lambda$}
        \[
          (r+\lambda)\,V = \frac{a}{2}(1-V)^2
        \]
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \begin{center}
    Optimal velocity in both cases: \quad $\kappa^* = a\,(1-V)$
  \end{center}
\end{frame}

% 9 ---------------------------------------------------------------
\begin{frame}{Result 1: velocity follows a square-root rule}
  \[
    \kappa^* \;\approx\; \sqrt{2a\,(r+\lambda)}
    \qquad\Longrightarrow\qquad
    \frac{\partial \ln \kappa^*}{\partial \ln (r+\lambda)} = \tfrac12
  \]
  \begin{center}
  \begin{tabular}{@{}lccccc@{}}
    \toprule
    Demurrage $\lambda$ (p.a.) & 0.5\% & 1\% & 2\% & 3\% & 5\% \\
    \midrule
    Velocity multiplier ($r=4\%$) & 1.06 & 1.12 & 1.22 & 1.32 & 1.49 \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.5em}
  \begin{itemize}
    \item Same elasticity as Baumol--Tobin money demand: the holding cost is just an interest rate
    \item Policy-sized rates move velocity \textbf{smoothly} by 10--30\%: no deadline, no spike
  \end{itemize}
\end{frame}

% 12 --------------------------------------------------------------
\begin{frame}{Why programmed holding costs must take the form of demurrage}
  \begin{center}
  \begin{tabular}{@{}lcc@{}}
    \toprule
    Property & Hard expiry & Demurrage \\
    \midrule
    Real waste (junk spending at $T$) & 2--27\% of issuance & none if $V>\phi$ \\
    Value of one unit & depends on vintage & identical for all units \\
    Fungibility / singleness of money & \alert{broken} & preserved \\
    Velocity path & rising, mass point at $T$ & constant \\
    Applicable to the whole money stock & no (needs vintages) & yes \\
    Is it an interest rate? & no & \textbf{yes}: $i=-\lambda$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.8em}
  \begin{center}
    \key{Expiry is a fiscal-programme tool. Demurrage is a monetary-policy instrument.}\\[0.3em]
    A central bank that programs a holding cost into money can only use the second.
  \end{center}
\end{frame}

% 13 --------------------------------------------------------------
\begin{frame}{Orders of magnitude: vouchers are not rates}
  \begin{itemize}
    \setlength\itemsep{0.8em}
    \item Matching voucher-strength velocity needs demurrage of \textbf{160\%--1{,}200\% per year}
    \item Policy-sized rates ($-0.5\%$ to $-3\%$) raise velocity by \textbf{6--32\%}
    \item Gap of \key{two to three orders of magnitude}
    \item Voucher effects also come from channels a rate does not have:
      \begin{itemize}
        \item transfer of new money (income effect)
        \item salience of a deadline \muted{(Boehm et al.\ 2024)}
        \item minimum-spend notches \muted{(Ding et al.\ 2024)}
      \end{itemize}
  \end{itemize}
  \vspace{0.5em}
  \begin{center}
    \key{The digital euro cannot be a stimulus voucher. At most it can carry a policy rate.}
  \end{center}
\end{frame}

% 14 --------------------------------------------------------------
\begin{frame}{Claim 3: can a negative digital-euro rate break the ELB?}
  Floor on deposit rates, with cash still in circulation:
  \[
    \underbrace{\max\{\,0,\; -c_{\text{cash}}\}=0}_{\text{Art.\ 16(8)}}
    \qquad\text{vs}\qquad
    \underbrace{\max\{-\lambda,\; -c_{\text{cash}}\} \ \ge\ -c_{\text{cash}}}_{\text{remunerated}}
  \]
  \begin{itemize}
    \setlength\itemsep{0.6em}
    \item \textbf{Asymmetry:} zero remuneration \emph{raises} the floor from $-c_{\text{cash}}$ to $0$;
          negative remuneration \emph{cannot} push it below $-c_{\text{cash}}$
    \item Going deeper needs cash reform: a crawling cash--reserve exchange rate \muted{(Agarwal \& Kimball 2019)}
    \item Other leaks: stablecoins, foreign currency, banks that do not pass rates on
  \end{itemize}
  \vspace{0.4em}
  \begin{center}
    \key{Remuneration flexibility protects existing policy space; it does not create new space.}
  \end{center}
\end{frame}

% 15 --------------------------------------------------------------
\begin{frame}{Policy proposal \muted{(for discussion)}}
  \begin{block}{Keep zero as the default, add a contingent rate clause}
    \begin{itemize}
      \item Normal times: $i_{DE}=0$, as in Art.\ 16(8) \muted{(trust, simplicity, no ``shrinking money'')}
      \item If the deposit facility rate is below zero, holdings above a tier threshold $M_1<\bar M$ pay
        \[
          i_{DE} = \max\{\text{DFR},\; -\bar c\}, \qquad \bar c \approx \text{cash holding cost}
        \]
      \item Implemented as \textbf{continuous demurrage}, never as expiry
    \end{itemize}
  \end{block}
  \vspace{0.4em}
  \begin{itemize}
    \item Tier 1 ($\le M_1$) protects households and the ``money'' role
    \item Tier 2 removes the frictionless-cash arbitrage at the margin
    \item Rate bounded by $-\bar c$: no incentive to flee into banknotes
  \end{itemize}
\end{frame}

% 16 --------------------------------------------------------------
\begin{frame}{Takeaways}
  \begin{enumerate}
    \setlength\itemsep{0.9em}
    \item A zero-rate digital euro is \key{cash without storage costs}: it hardens the lower bound on the balances the limit allows
    \item Programmed holding costs must be \key{demurrage}: expiry wastes resources, breaks fungibility and is not a rate
    \item Vouchers and rates differ by \key{two to three orders of magnitude}: the digital euro cannot be a stimulus voucher
    \item Remuneration flexibility \key{protects} policy space; only cash reform \key{creates} it
  \end{enumerate}
  \vspace{1em}
  \muted{Companion talk: how large $\bar M$ should be. This talk: what price should sit under $\bar M$.}
\end{frame}

\begin{frame}{B1. Derivations}
  \small
  \textbf{Holder's problem.} Per unit of balance, with search cost $\kappa^2/2a$ and value $V$:
  \[
    (r+\lambda)V = \max_{\kappa}\Bigl\{\kappa(1-V) - \frac{\kappa^2}{2a}\Bigr\}
    \;\Rightarrow\; \kappa^*=a(1-V),\quad (r+\lambda)V=\frac a2(1-V)^2 .
  \]
  \textbf{Square-root rule.} Let $x=1-V$ (small when $r+\lambda \ll a$):
  $(r+\lambda)(1-x)=\tfrac a2 x^2 \Rightarrow x\approx\sqrt{2(r+\lambda)/a}$, so $\kappa^*=ax\approx\sqrt{2a(r+\lambda)}$.

  \textbf{Expiry.} With remaining time $\tau$, the HJB becomes the ODE
  $\;dV/d\tau=-rV+\tfrac a2(1-V)^2$, \; $V(0)=\phi$.
  $V$ rises above $\phi$ immediately if $\tfrac a2(1-\phi)^2>r\phi$, so dumping happens only at $T$.
  Velocity $\kappa(\tau)=a(1-V(\tau))\uparrow a(1-\phi)$ as $\tau\to0$: \textbf{finite}, but the remaining balance is a mass point at $T$.

  \textbf{Secondary-market discount.} An expiring unit trades at $V(\tau)$, so the discount $1-V(\tau)$ rises from about $1-V(T)$ to $1-\phi$; units of different vintages trade at different prices.

  \textbf{Calibration.} $\lambda$ is chosen so that $\kappa_{\text{demurrage}}$ equals the time-average of $\kappa_{\text{expiry}}(t)$ over $[0,T]$.
\end{frame}

% B3 --------------------------------------------------------------
\begin{frame}{B3. Limitations}
  \begin{itemize}
    \setlength\itemsep{0.6em}
    \item Partial equilibrium: no prices, no general-equilibrium feedback from extra spending
    \item Search technology is stylised; $a$ and $\phi$ are not estimated
    \item The ``cliff after $T$'' is not modelled; evidence is mixed: Xing et al.\ (2023) find purchases are moved forward, Ding et al.\ (2024) find no short-run intertemporal substitution
    \item Behavioural channels (salience, notches) are outside the model and favour vouchers
    \item $c_{\text{cash}}$ is not estimated; the ELB literature gives a range rather than a number
  \end{itemize}
\end{frame}
