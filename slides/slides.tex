        \setlength\itemsep{0.8em}

        \item A programmable currency can charge for holding it
        \item \textbf{Expiry:} the whole cost falls on day $T$

% 4 ---------------------------------------------------------------
\begin{frame}{To compare them, take a holder who searches for good uses of money and can always dump it at a loss}
  \begin{itemize}
    \item A unit of money is worth $V$ to its holder; a good purchase is worth $1$
    \item Searching at intensity $\kappa$ costs $\kappa^2/2a$; $\kappa$ is the \textbf{velocity}
    \item A bad use (junk goods, selling at a discount) is always available at $\phi<1$
  \end{itemize}
  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Expiry at $T$ \quad ($\tau = T-t$ left)}
        \[
          \frac{dV}{d\tau} = -rV + \frac{a}{2}(1-V)^2,\quad V(0)=\phi
        \]
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Decay at rate $\lambda$}
        \[
          (r+\lambda)\,V = \frac{a}{2}(1-V)^2
        \]
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \begin{center}
    In both cases the holder spends at speed \quad $\kappa^* = a\,(1-V)$:\quad the less money is worth holding, the faster it moves.
  \end{center}
\end{frame}

% 6 ---------------------------------------------------------------
\begin{frame}{Continuous decay keeps every euro equal, and it is nothing more than a negative interest rate}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \small
      \begin{tabular}{@{}lcc@{}}
        \toprule
        & Expiry & Decay \\
        \midrule
        Fire sale at the deadline & 3--55\% & none \\
        Value of one unit & by vintage & all equal \\
        One euro = one euro? & \alert{no} & yes \\
        Applies to all money & no & yes \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.44\textwidth}
      Decay at $\lambda$ is the rate $i=-\lambda$. Velocity follows
      \[
        \kappa^* \approx \sqrt{2a\,(r+\lambda)}
      \]
      the square-root rule of Baumol--Tobin money demand.
    \end{column}
  \end{columns}
  \vspace{1em}
  \begin{center}
    \key{Expiry is a feature of a spending programme. Decay can be a feature of money.}
  \end{center}
\end{frame}

% 8 ---------------------------------------------------------------
\begin{frame}{So the real question is the digital euro's interest rate, and the draft regulation fixes it at zero}
  \vspace{2em}
  \begin{center}
    {\Large \emph{``The digital euro shall not bear interest.''}}\\[0.5em]
    \muted{Commission proposal for a digital euro regulation, Art.\ 16(8), June 2023}
  \end{center}
  \vspace{1.5em}
  \begin{itemize}
    \setlength\itemsep{0.6em}
    \item Not positive, not negative, in any circumstances
    \item Paired with a holding limit, currently discussed at \euro 1{,}500--3{,}000 per person
    \item Council and Parliament are negotiating the final text now
  \end{itemize}
\end{frame}

% 9 ---------------------------------------------------------------
\begin{frame}{A zero-rate digital euro is cash without storage costs, which lifts the lower bound back to zero}
  Depositors can always switch to central-bank money, so banks cannot pay less than
  \[
    r_D \;\ge\; \max_j\,\{\, i_j - c_j \,\} \qquad (i_j \text{: rate}, \; c_j \text{: cost of holding})
  \]
  \begin{center}
  \begin{tabular}{@{}lccc@{}}
    \toprule
    & rate & holding cost & floor for deposit rates \\
    \midrule
    Cash & $0$ & storage, insurance $c>0$ & $-c$ \\
    Digital euro at zero & $0$ & $\approx 0$ & \alert{$0$} \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.3em}
  \begin{itemize}
    \item 2019--22: the ECB went to $-0.50\%$ because hoarding banknotes is costly
    \item Next time, every holder can move up to the limit into a costless zero-rate asset
    \item Deposits leave banks exactly when negative rates already squeeze their margins
  \end{itemize}
\end{frame}

% 10 --------------------------------------------------------------
\begin{frame}{A negative digital-euro rate can win that space back, but no more while cash still exists}
  \vspace{0.5em}
  \[
    \text{floor} \;=\;
    \underbrace{\max\{\,0,\; -c\,\}=0}_{\text{zero rate}}
    \qquad\longrightarrow\qquad
    \underbrace{\max\{-\lambda,\; -c\,\}\;\ge\;-c}_{\text{decay at }\lambda}
  \]
  \vspace{0.5em}
  \begin{itemize}
    \setlength\itemsep{0.8em}
    \item A zero rate \textbf{raises} the floor from $-c$ to $0$
    \item Decay can bring it back down to $-c$, but \textbf{not below}: beyond that, people switch to banknotes
    \item Going deeper requires repricing cash itself, e.g.\ a crawling cash--reserve exchange rate
  \end{itemize}
  \vspace{0.3em}
  \begin{center}
    \key{A flexible rate protects the policy space we have; it cannot create more.}
  \end{center}
  \source{Cash repricing: Agarwal \& Kimball (IMF WP 19/84, 2019).}
\end{frame}

% 11 --------------------------------------------------------------
\begin{frame}{The best design keeps zero as the default and lets balances above a threshold decay when policy rates turn negative}
  \begin{block}{A contingent rate clause}
    \[
      i_{DE} =
      \begin{cases}
        0 & \text{on all balances, when DFR} \ge 0\\[0.2em]
        0 & \text{on balances up to } M_1\\[0.2em]
        \max\{\text{DFR},\,-\bar c\} & \text{on balances above } M_1, \text{ when DFR}<0
      \end{cases}
    \]
  \end{block}
  \vspace{0.1em}
  \begin{itemize}
    \setlength\itemsep{0.35em}
    \item \textbf{Zero in normal times:} simple, trusted, never ``money that melts''
    \item \textbf{Small balances never decay:} households and payments are protected
    \item \textbf{Decay, never expiry:} every euro stays equal to every other
    \item \textbf{Floored at $-\bar c$:} no reason to flee into banknotes
  \end{itemize}
\end{frame}

% 12 --------------------------------------------------------------
\begin{frame}{The digital euro should be able to bear interest: only as decay, and only when rates are negative}
  \vspace{1.5em}
  \begin{enumerate}
    \setlength\itemsep{1em}
    \item Deadlines make people spend, but an expiry date belongs to a spending programme, not to money
    \item The only holding cost money can carry is continuous decay, which is simply a negative interest rate
    \item Fixing that rate at zero turns the digital euro into costless cash and gives up policy space
    \item A contingent, tiered rate keeps zero for normal times and protects that space when it matters
  \end{enumerate}
\end{frame}

\begin{frame}{B1. Derivations}
  \small
  \textbf{Holder's problem.} Per unit of balance:
  \[
    (r+\lambda)V = \max_{\kappa}\Bigl\{\kappa(1-V) - \frac{\kappa^2}{2a}\Bigr\}
    \;\Rightarrow\; \kappa^*=a(1-V),\quad (r+\lambda)V=\frac a2(1-V)^2 .
  \]
  \textbf{Square-root rule.} With $x=1-V$ small: $(r+\lambda)(1-x)=\tfrac a2 x^2 \Rightarrow x\approx\sqrt{2(r+\lambda)/a}$, so $\kappa^*\approx\sqrt{2a(r+\lambda)}$.

  \textbf{Expiry.} With $\tau$ left, $dV/d\tau=-rV+\tfrac a2(1-V)^2$, $V(0)=\phi$. If $\tfrac a2(1-\phi)^2>r\phi$, $V$ rises above $\phi$ at once, so dumping happens only at $T$.
  Velocity $\kappa(\tau)=a(1-V(\tau))$ rises to $a(1-\phi)$: finite, but the unspent balance is spent all at once at $T$.

  \textbf{Secondary market.} An expiring unit trades at $V(\tau)$; units of different ages trade at different prices.

  \textbf{Calibration.} $\lambda$ set so that decay velocity equals the average expiry velocity over $[0,T]$.
\end{frame}
